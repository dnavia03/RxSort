class PillSorterApp(ctk.CTk):
def __init__(self):
super().__init__()
ctk.set_appearance_mode("Dark")
ctk.set_default_color_theme("blue")
self.title(WINDOW_TITLE)
self.geometry("900x520")
self.minsize(760, 480)
# Communicate between the hardware worker and GUI thread.
self.ui_queue = queue.Queue()
self.sorting_in_progress = False
self.buttons = []
self.create_layout()
# Process GUI updates on the main thread.
self.after(100, self.process_ui_queue)
# ========================================================
# GUI LAYOUT
# ========================================================

self.grid_columnconfigure(1, weight=1)
self.grid_rowconfigure(0, weight=1)
# ---------------- LEFT: CONTROL PANEL ----------------
self.control_panel = ctk.CTkFrame(
self,
corner_radius=15,
)

padx=20,
pady=20,
sticky="nsew",
)
self.title_lbl = ctk.CTkLabel(
self.control_panel,
text="Carousel Control Hub",
font=("Arial", 18, "bold"),
)
self.title_lbl.pack(pady=20)
self.status_lbl = ctk.CTkLabel(
self.control_panel,
text="System Ready",
text_color="#2ecc71",
font=("Arial", 14, "bold"),
)

# Create one button for every configured slot.
for slot_id, data in PILL_DATABASE.items():
button = ctk.CTkButton(
self.control_panel,

f"Dispense {data['name'].split()[0]} "

pady=12,

self.buttons.append(button)
self.log_lbl = ctk.CTkLabel(
self.control_panel,

)

# ---------------- RIGHT: RESULTS PANEL ----------------
self.result_panel = ctk.CTkFrame(
self,
corner_radius=15,

)

)

self.image_display = ctk.CTkLabel(
self.result_panel,

)

self.pill_name_lbl = ctk.CTkLabel(
self.result_panel,

# ========================================================
# CAROUSEL CONTROL
# ========================================================
def start_sorting_routine(self, slot_id):
"""Start the hardware routine without freezing the GUI."""
if self.sorting_in_progress:
return
if slot_id not in PILL_DATABASE or slot_id not in SLOT_ANGLES:
self.log_lbl.configure(text="Error: Invalid slot configuration.")
return
self.sorting_in_progress = True
for button in self.buttons:
button.configure(state="disabled")
self.status_lbl.configure(
text="Rotating Carousel...",
text_color="#f39c12",
)
self.log_lbl.configure(
text=f"Logs: Moving to Slot {slot_id}..."
)
worker = threading.Thread(
target=self.run_hardware_carousel,
args=(slot_id,),
daemon=True,
)
worker.start()
def run_hardware_carousel(self, slot_id):
"""Run hardware operations in a background thread."""
try:
target_angle = SLOT_ANGLES[slot_id]
# Move the carousel if a compatible servo is configured.
if carousel_servo is not None:
carousel_servo.angle = target_angle
# Simulate motor travel time.
time.sleep(2.0)
# Send results to the main GUI thread.
self.ui_queue.put(
("success", slot_id, target_angle)
)
except Exception as error:
self.ui_queue.put(
("error", slot_id, str(error))
)
# ========================================================
# GUI THREAD UPDATES
# ========================================================
def process_ui_queue(self):
"""Process background-thread messages safely."""
try:
while True:
message = self.ui_queue.get_nowait()
event_type = message[0]
slot_id = message[1]
if event_type == "success":
target_angle = message[2]
self.log_lbl.configure(
text=(
f"Logs: Slot {slot_id} "
f"positioned at {target_angle} degrees."
)
)
self.display_pill_results(slot_id)
self.status_lbl.configure(
text="System Ready",
text_color="#2ecc71",
)
elif event_type == "error":
error_message = message[2]
self.log_lbl.configure(
text=f"Hardware Error: {error_message}"
)
self.status_lbl.configure(
text="Hardware Error",
text_color="#e74c3c",
)
self.sorting_in_progress = False
for button in self.buttons:
button.configure(state="normal")
except queue.Empty:
pass
# Schedule the next queue check on the main thread.
self.after(100, self.process_ui_queue)
# ========================================================
# PILL INFORMATION DISPLAY
# ========================================================
def display_pill_results(self, slot_id):
"""Display the selected pill's image and information."""
pill_data = PILL_DATABASE[slot_id]
image_path = pill_data["image"]
if os.path.exists(image_path):
try:
with Image.open(image_path) as source_image:
pil_img = source_image.copy()
ctk_img = ctk.CTkImage(
light_image=pil_img,
dark_image=pil_img,
size=(250, 180),
)
# Keep a reference to prevent the image disappearing.
self.current_pill_image = ctk_img
self.image_display.configure(
image=ctk_img,
text="",
)
except Exception as error:
self.image_display.configure(
image=None,
text=f"Error loading image:
{error}",
)
else:
self.image_display.configure(
image=None,
text=f"Missing image asset:
{image_path}",
)
self.pill_name_lbl.configure(
text=f"Name: {pill_data['name']}"
)
self.pill_dose_lbl.configure(
text=f"Dosage: {pill_data['dosage']}"
)
self.pill_info_lbl.configure(
text=f"Instructions: {pill_data['instructions']}"
)
